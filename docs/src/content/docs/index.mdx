import { CardGrid, LinkCard } from '@astrojs/starlight/components';

## What it does

<CardGrid>
  <LinkCard
    title="Inject the Twilio client"
    href="/configuration/"
    description="Register the SDK client once with TwilioModule.forRoot(), then inject it anywhere with @InjectTwilio() or the TwilioService wrapper."
  />
  <LinkCard
    title="Validate webhook signatures"
    href="/features/webhook-validation/"
    description="TwilioWebhookGuard and @TwilioWebhook() verify the X-Twilio-Signature header, including behind proxies and for JSON bodies."
  />
  <LinkCard
    title="Return TwiML responses"
    href="/features/twiml-responses/"
    description="Return a VoiceResponse, MessagingResponse or FaxResponse builder from a controller and TwimlInterceptor serializes it to XML."
  />
  <LinkCard
    title="Handle Twilio errors"
    href="/features/error-handling/"
    description="TwilioExceptionFilter maps the SDK's RestException onto a Nest HttpException, preserving the Twilio error code and more_info link."
  />
  <LinkCard
    title="Use multiple accounts"
    href="/features/multi-account/"
    description="Register named clients for subaccounts with TwilioModule.registerClient() and inject them with @InjectTwilio(name)."
  />
  <LinkCard
    title="Mint access tokens"
    href="/features/access-tokens/"
    description="Issue short-lived tokens for the Voice, Video, Conversations and Sync client SDKs, signed with an API key."
  />
  <LinkCard
    title="Add health checks"
    href="/features/health-checks/"
    description="Report Twilio API reachability and credential validity through a Terminus health indicator."
  />
  <LinkCard
    title="Test your integration"
    href="/guides/testing/"
    description="Override the injected client, sign fixture webhooks, and assert on TwiML and mapped errors without calling Twilio."
  />
</CardGrid>
